dinamico:

ciclico:
mpicc -std=c11 -O3 -Wall -Wextra -pedantic -Iinclude \
  src/bruteforce_mpi_cyclic.c  src/des_utils.c -lcrypto -o bin/bruteforce_mpi_cyclic

  -o bin/bruteforce_mpi_permuted

  -o bin/bruteforce_mpi_dynamic_adaptive
